/**
 * La intro: una pantalla que se abre sobre toda la página al entrar (cada vez o una vez por visita;
 * logo, nombre letra por letra y una frase) y se levanta como un telón. Cada tema la viste con sus reglas
 * `.pv-<tema> .pv-intro-*`. Se saltea con un toque y no aparece si el visitante pidió menos movimiento.
 * En el editor se ve igual que publicada: al cargar, al cambiar sus propiedades y con «Ver la intro»,
 * desde una franja angosta que marca dónde está la sección.
 */
export function Intro({ seccion }: PropsSeccion) {
  const p = propsDe<Props>(seccion)
  const { local } = useLocal()
  const { isPreview } = useSitio()
  const { tema } = useSitioPagina()
  const quieto = useReducedMotion()
  // Cómo se va: un telón que sube o, en los temas con círculo, un agujero que se abre desde el centro.
  const salida = SALIDAS[tema.plantilla as keyof typeof SALIDAS] ?? SALIDAS.telon
  const clave = `piru-intro:${local?.nombre ?? ""}`
  const [abierta, setAbierta] = useState(() => {
    if (quieto) return false
    if (isPreview || p.repetir !== "sesion") return true
    try {
      return !sessionStorage.getItem(clave)
    } catch {
      return true
    }
  })
  const ms = DURACION[p.duracion ?? "normal"] ?? DURACION.normal
  const firma = JSON.stringify(seccion.props ?? {})

  // En el editor, cambiar una propiedad la vuelve a mostrar con el cambio.
  useEffect(() => {
    if (isPreview && !quieto) setAbierta(true)
  }, [firma, isPreview, quieto])

  useEffect(() => {
    if (!abierta) return
    if (!isPreview) {
      try {
        sessionStorage.setItem(clave, "1")
      } catch {
        /* sin almacenamiento: se ve igual */
      }
    }
    const previo = document.documentElement.style.overflow
    document.documentElement.style.overflow = "hidden"
    const t = window.setTimeout(() => setAbierta(false), ms)
    return () => {
      window.clearTimeout(t)
      document.documentElement.style.overflow = previo
    }
  }, [abierta, ms, firma, clave, isPreview])

  if (!local) return null

  const pantalla = (
    <AnimatePresence>
      {abierta && (
        <motion.div
          key={firma}
          className="pv-intro"
          role="presentation"
          onClick={() => setAbierta(false)}
          initial={salida.inicial}
          exit={salida.salida}
          transition={{ duration: salida.duracion, ease: [0.76, 0, 0.24, 1] }}
        >
          <Escena p={p} ms={ms} />
          <span className="pv-intro-saltar">Tocá para entrar</span>
        </motion.div>
      )}
    </AnimatePresence>
  )

  if (!isPreview) return pantalla
  return (
    <Seccion seccion={seccion} etiqueta="Intro" className="pv-intro-franja">
      <span>Intro · se ve al entrar a la página</span>
      <button type="button" onClick={() => setAbierta(true)}>
        Ver la intro
      </button>
      {pantalla}
    </Seccion>
  )
}

function Escena({ p, ms }: { p: Partial<Props>; ms: number }) {
  const { local } = useLocal()
  const { contenido, tema } = useSitioPagina()
  if (!local) return null
  // En Tarascón cada letra sube estirada y rebota, como un mordisco (ver `.pv-tarascon .pv-intro-nombre`).
  const muerde = tema.plantilla === "tarascon"
  // En Ñam cada letra cae desde arriba como una figurita que se pega torcida (ver `.pv-nam .pv-intro-nombre`).
  const cae = tema.plantilla === "nam"
  // En Chispa cada letra salta encendida desde abajo, como una chispa de la plancha (ver `.pv-chispa .pv-intro-nombre`).
  const chispa = tema.plantilla === "chispa"
  // En Sorbo cada letra baja colgando como un fideo y queda en su escalón (ver `.pv-sorbo .pv-intro-nombre`).
  const sorbo = tema.plantilla === "sorbo"
  // En Girasol cada letra amanece desde abajo, girando y con desenfoque, por delante del sol (ver `.pv-girasol .pv-intro-nombre`).
  const girasol = tema.plantilla === "girasol"
  // En Calcomanía cada letra cae grande y se pega de un golpe, torcida como un recorte (ver `.pv-calcomania .pv-intro-nombre`).
  const pega = tema.plantilla === "calcomania"
  // En Baobab cada letra cae estirada y se asienta con un rebote, como el titular que pisa fuerte (ver `.pv-baobab .pv-intro-nombre`).
  const pisa = tema.plantilla === "baobab"
  // En Porción cada letra entra girando sobre el arco de sabores, como la rueda de la portada (ver `.pv-porcion .pv-intro-nombre`).
  const rueda = tema.plantilla === "porcion"
  // En Tazón cada letra se pinta de izquierda a derecha como un trazo de pincel seco (ver `.pv-tazon .pv-intro-nombre`).
  const trazo = tema.plantilla === "tazon"
  // En Kuro cada letra se asienta como tinta: grande y desenfocada, se achica a su lugar (ver `.pv-kuro .pv-intro-nombre`).
  const asienta = tema.plantilla === "kuro"
  // En Hanko cada letra se estampa como un sello: llega grande y desenfocada y se asienta en el papel (ver `.pv-hanko .pv-intro-nombre`).
  const sello = tema.plantilla === "hanko"
  // En Sumi cada letra aparece como tinta que se asienta en el papel: baja desenfocada y se aclara (ver `.pv-sumi .pv-intro-nombre`).
  const tinta = tema.plantilla === "sumi"
  // En Miga cada letra leva como masa: nace chiquita, se infla con un rebote y queda bailando torcida (ver `.pv-miga .pv-intro-nombre`).
  const leva = tema.plantilla === "miga"
  // En Mordisco cada letra cae de golpe desde arriba, gigante y torcida, y se estampa como un afiche pegado a las apuradas (ver `.pv-mordisco .pv-intro-nombre`).
  const golpe = tema.plantilla === "mordisco"
  // En Damero cada letra sube de golpe desde abajo, estirada y torcida, y rebota como un cartel que se clava (ver `.pv-damero .pv-intro-nombre`).
  const clava = tema.plantilla === "damero"
  const oscuro = esOscuro(tema.fondo)
  const logo = p.mostrarLogo !== false ? (oscuro ? contenido.logo : contenido.logoClaro) ?? logoPara(local, oscuro) : null
  const palabras = local.nombre.split(/\s+/).filter(Boolean)
  let indice = 0
  return (
    <div className="pv-intro-escena" style={{ "--pv-intro-ms": `${ms}ms` } as CSSProperties}>
      <div className="pv-intro-brillo" aria-hidden="true" />
      {logo && (
        <motion.img
          className="pv-intro-logo"
          src={logo}
          alt=""
          initial={{ opacity: 0, scale: 0.8, filter: "blur(10px)" }}
          animate={{ opacity: 1, scale: 1, filter: "blur(0px)" }}
          transition={{ duration: 0.9, ease: SUAVE }}
        />
      )}
      {p.antetitulo?.trim() && (
        <motion.p className="pv-intro-antetitulo" data-piru-prop="antetitulo" initial={{ opacity: 0, letterSpacing: "0.6em" }} animate={{ opacity: 1, letterSpacing: "0.32em" }} transition={{ duration: 1.1, delay: 0.2, ease: SUAVE }}>
          {p.antetitulo}
        </motion.p>
      )}
      <h2 className="pv-intro-nombre" aria-label={local.nombre}>
        {palabras.map((palabra, w) => (
          <span key={w} className="pv-intro-palabra" aria-hidden="true">
            {[...palabra].map((letra) => {
              const i = indice++
              return (
                <motion.span
                  key={i}
                  style={{ "--n": i } as CSSProperties}
                  initial={leva ? { y: "45%", scale: 0.15, rotate: i % 2 ? 16 : -16, opacity: 0 } : golpe ? { y: "-80%", scale: 2.4, rotate: i % 2 ? 16 : -16, opacity: 0 } : clava ? { y: "130%", scaleY: 1.7, rotate: i % 2 ? 7 : -7, opacity: 0 } : asienta ? { scale: 1.6, y: "-12%", opacity: 0, filter: "blur(14px)" } : sello ? { y: "18%", scale: 1.25, opacity: 0, filter: "blur(14px)" } : tinta ? { y: "-40%", scale: 1.4, opacity: 0, filter: "blur(14px)" } : trazo ? { x: "-60%", skewX: -25, scaleX: 1.6, opacity: 0, filter: "blur(8px)" } : rueda ? { y: "60%", x: "-40%", rotate: -70, opacity: 0, filter: "blur(6px)" } : pisa ? { y: "-130%", scaleY: 1.9, opacity: 0 } : pega ? { scale: 2.6, rotate: i % 2 ? 24 : -24, opacity: 0, filter: "blur(8px)" } : girasol ? { y: "90%", rotate: i % 2 ? 20 : -20, scale: 0.5, opacity: 0, filter: "blur(12px)" } : sorbo ? { y: "-140%", scaleY: 2.2, opacity: 0 } : chispa ? { y: "70%", scale: 0.2, rotate: i % 2 ? 12 : -12, opacity: 0 } : cae ? { y: "-160%", rotate: i % 2 ? 18 : -18, opacity: 0 } : muerde ? { y: "110%", scaleY: 1.8, opacity: 0 } : { y: "105%", opacity: 0 }}
                  animate={{ y: "0%", ...(clava ? { rotate: leva ? (i % 2 ? 4 : -4) : 0 } : {}), ...(rueda ? { x: "0%", filter: "blur(0px)" } : {}), ...(trazo ? { x: "0%", skewX: 0, scaleX: 1, filter: "blur(0px)" } : {}), ...(tinta || sello || asienta ? { filter: "blur(0px)" } : {}), rotate: golpe ? (i % 3) - 1 : cae ? (i % 2 ? 3 : -3) : pega ? ((i % 3) - 1) * 5 : 0, scale: 1, scaleY: 1, opacity: 1, ...(girasol || pega ? { filter: "blur(0px)" } : {}) }}
                  transition={leva ? { type: "spring", bounce: 0.6, duration: 0.8, delay: 1.05 + i * 0.06 } : golpe ? { type: "spring", bounce: 0.5, duration: 0.6, delay: 0.75 + i * 0.07 } : clava ? { type: "spring", bounce: 0.5, duration: 0.7, delay: 0.95 + i * 0.05 } : asienta ? { duration: 1, delay: 0.85 + i * 0.07, ease: SUAVE } : sello ? { duration: 1.3, delay: 0.9 + i * 0.08, ease: SUAVE } : tinta ? { duration: 0.9, delay: 1.1 + i * 0.06, ease: SUAVE } : trazo ? { type: "spring", bounce: 0.3, duration: 0.75, delay: 0.8 + i * 0.055 } : rueda ? { type: "spring", bounce: 0.35, duration: 0.8, delay: 0.9 + i * 0.06 } : pisa ? { type: "spring", bounce: 0.4, duration: 0.7, delay: 0.6 + i * 0.05 } : pega ? { type: "spring", bounce: 0.35, duration: 0.5, delay: 0.7 + i * 0.08 } : girasol ? { type: "spring", bounce: 0.45, duration: 0.9, delay: 0.75 + i * 0.065 } : sorbo ? { type: "spring", bounce: 0.55, duration: 0.8, delay: 0.55 + i * 0.07 } : chispa ? { type: "spring", bounce: 0.5, duration: 0.65, delay: 0.95 + i * 0.055 } : cae ? { type: "spring", bounce: 0.55, duration: 0.75, delay: 0.5 + i * 0.06 } : muerde ? { type: "spring", bounce: 0.45, duration: 0.7, delay: 0.45 + i * 0.07 } : { duration: 0.7, delay: 0.35 + i * 0.045, ease: SUAVE }}
                >
                  {letra}
                </motion.span>
              )
            })}
          </span>
        ))}
      </h2>
      <motion.span className="pv-intro-linea" aria-hidden="true" initial={{ scaleX: 0 }} animate={{ scaleX: 1 }} transition={{ duration: 1, delay: 0.7, ease: SUAVE }} />
      {p.frase?.trim() && (
        <motion.p className="pv-intro-frase" data-piru-prop="frase" initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.8, delay: 1, ease: SUAVE }}>
          {p.frase}
        </motion.p>
      )}
    </div>
  )
}
