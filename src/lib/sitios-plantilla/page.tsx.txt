import { useEffect, useMemo, useState } from "react"
import { AnimatePresence, motion, useReducedMotion } from "motion/react"
import { ShoppingBag } from "lucide-react"
import { useCarrito, useCarta, useLocal, useSitio, type Producto } from "@piru/tienda"
import { CONTENIDO } from "./contenido"
import "./style.css"
import "./fuentes.css"
import { SitioContexto, temaDe, useFuentes, variablesTema } from "./components/Sitio"
import { SECCIONES } from "./secciones"
import { Pedido } from "./components/Pedido"
import { DetalleProducto } from "./components/DetalleProducto"
import { Checkout } from "./components/Checkout"

/*
 * Tienda de Páginas armada por secciones.
 *
 * La página es la lista `CONTENIDO.secciones` (en `contenido.ts`): cada una tiene un `tipo`, si
 * está visible y sus propiedades. El editor visual sólo cambia esos datos; el diseño de cada
 * sección vive en `secciones/<Tipo>.tsx` y se elige en `secciones/index.ts`.
 *
 * Lo que no es una sección queda siempre: la barra del pedido, el detalle de producto
 * (`components/DetalleProducto.tsx`) y el checkout de Piru (`components/Checkout.tsx`).
 */
export default function Page() {
  const { estado } = useLocal()
  const { productos } = useCarta()
  const { isPreview } = useSitio()
  const carrito = useCarrito()
  const quieto = useReducedMotion()
  const contenido = CONTENIDO
  const tema = temaDe(contenido)
  useFuentes(contenido, tema)
  const [pedidoAbierto, setPedidoAbierto] = useState(false)
  // El detalle de producto es parte del diseño (`components/DetalleProducto.tsx`), no del runtime.
  const [detalle, setDetalle] = useState<Producto | null>(null)
  const secciones = useMemo(() => (contenido.secciones ?? []).filter((s) => s.visible !== false), [contenido.secciones])
  const contexto = useMemo(() => ({ contenido, tema, secciones, abrirProducto: setDetalle }), [contenido, tema, secciones])

  useEffect(() => {
    if (carrito.count === 0) setPedidoAbierto(false)
  }, [carrito.count])

  if (estado === "cargando") {
    return (
      <div className="pv-cargando" style={variablesTema(tema)} aria-busy="true" aria-label="Cargando la tienda">
        <span />
      </div>
    )
  }
  if (estado === "error") {
    return (
      <main className="pv-estado" style={variablesTema(tema)}>
        <h1>No pudimos cargar la tienda</h1>
        <p>Probá de nuevo en un rato.</p>
      </main>
    )
  }

  return (
    <main className={`pv-sitio pv-${tema.plantilla}`} style={variablesTema(tema)}>
      <SitioContexto.Provider value={contexto}>
        {secciones.map((seccion) => {
          const Componente = SECCIONES[seccion.tipo]
          if (Componente) return <Componente key={seccion.id} seccion={seccion} />
          return isPreview ? (
            <p key={seccion.id} className="pv-vacia-texto" data-piru-seccion={seccion.id}>
              No hay un componente para la sección «{seccion.tipo}». Agregalo en secciones/index.ts.
            </p>
          ) : null
        })}

        <AnimatePresence>
          {carrito.count > 0 && !pedidoAbierto && (
            <motion.button
              type="button"
              className="pv-barra"
              onClick={() => setPedidoAbierto(true)}
              initial={quieto ? false : { y: 80, opacity: 0 }}
              animate={{ y: 0, opacity: 1 }}
              exit={{ y: 80, opacity: 0 }}
            >
              <span className="pv-barra-cuenta">
                <ShoppingBag size={18} aria-hidden="true" /> {carrito.count}
              </span>
              <span>Ver pedido</span>
              <span>{carrito.totalFormatted}</span>
            </motion.button>
          )}
        </AnimatePresence>

        {pedidoAbierto && <Pedido onCerrar={() => setPedidoAbierto(false)} />}

        <DetalleProducto
          producto={detalle}
          estilo={contenido.detalleProducto}
          hermanos={productos}
          onNavegar={setDetalle}
          onCerrar={() => setDetalle(null)}
        />

        {/* El checkout vive en el runtime: acá sólo se elige cómo se ve y se activa con "Pedir". */}
        <Checkout
          variante={contenido.checkout ?? "pasos"}
          estilo={{
            acento: tema.primario,
            texto: tema.texto,
            superficie: tema.superficie,
            linea: tema.linea,
            radio: tema.plantilla === "moderna" ? 22 : tema.plantilla === "elegante" ? 4 : 16,
            fuente: tema.cuerpo,
          }}
        />
      </SitioContexto.Provider>
    </main>
  )
}
